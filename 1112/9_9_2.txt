8. Irvine32 라이브러리의 Str_ucase 프로시저를 수정하여 모든 문자를 소문자로 변경하도록 하는 방법을 보이시오.
정답:
Str_lcase PROC USES eax esi,
    pString:PTR BYTE
    mov esi, pString
L1:
    mov al, [esi]
    cmp al, 0
    je L3
    cmp al, 'A'      ; 'a' 대신 'A'와 비교
    jb L2
    cmp al, 'Z'      ; 'z' 대신 'Z'와 비교
    ja L2
    or BYTE PTR [esi], 00100000b ; 대문자를 소문자로 변환 (AND 11011111b 대신 OR 사용)
L2:
    inc esi
    jmp L1
L3:
    ret
Str_lcase ENDP

9. Str_trim 프로시저의 64비트 버전을 작성하시오.
정답:
; RCX = 문자열의 주소, RDX(DL) = 제거할 구분자 문자
Str_trim64 PROC USES rax rcx rdi
    mov rdi, rcx
    call Str_length64    ; 문자열 길이를 RAX에 반환한다고 가정
    cmp rax, 0
    je L3
    mov rcx, rax
    dec rax
    add rdi, rax         ; RDI가 문자열의 마지막 문자를 가리키도록 설정
L1:
    mov al, [rdi]
    cmp al, dl
    jne L2
    dec rdi
    loop L1
L2:
    mov BYTE PTR [rdi + 1], 0 ; 널 종료 문자 삽입
L3:
    ret
Str_trim64 ENDP
